```ts
  await page.screenshot({ path: testInfo.outputPath('boot.png') });
```

- [ ] **Step 5: Write `src/render/stage.ts`**

Every value and call order below mirrors the tuner so tuned settings render identically.

```ts
import * as THREE from 'three';
import { RoomEnvironment } from 'three/addons/environments/RoomEnvironment.js';
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';
import { OutputPass } from 'three/addons/postprocessing/OutputPass.js';

export interface LightSettings {
  keyIntensity: number; keyX: number; keyY: number; keyZ: number;
  fillIntensity: number; fillColor: string;
  rimIntensity: number; rimX: number; rimY: number; rimZ: number;
  envIntensity: number; exposure: number;
}

export interface BloomSettings {
  bloom: boolean; bloomThreshold: number; bloomStrength: number; bloomRadius: number;
}

export const BACKGROUND = '#070b1a';

export class Stage {
  readonly renderer: THREE.WebGLRenderer;
  readonly scene = new THREE.Scene();
  readonly camera = new THREE.PerspectiveCamera(26, 1, 0.1, 100);
  readonly key = new THREE.DirectionalLight(0xffffff, 1);
  readonly fill = new THREE.DirectionalLight(0xffffff, 1);
  readonly rim = new THREE.DirectionalLight(0xffffff, 1);
  private readonly composer: EffectComposer;
  private readonly bloom: UnrealBloomPass;

  constructor(container: HTMLElement) {
    this.renderer = new THREE.WebGLRenderer({ antialias: true });
    this.renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    container.append(this.renderer.domElement);

    this.scene.background = new THREE.Color(BACKGROUND);
    const pmrem = new THREE.PMREMGenerator(this.renderer);
    this.scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
    pmrem.dispose();

    this.camera.position.set(0, 0.3, 6.5);
    this.scene.add(this.key, this.fill, this.rim);

    this.composer = new EffectComposer(this.renderer);
    this.composer.addPass(new RenderPass(this.scene, this.camera));
    this.bloom = new UnrealBloomPass(new THREE.Vector2(1, 1), 0.35, 0.4, 0.8);
    this.composer.addPass(this.bloom);
    this.composer.addPass(new OutputPass());

    this.resize();
    addEventListener('resize', () => this.resize());
  }

  applyLighting(l: LightSettings): void {
    this.key.intensity = l.keyIntensity;
    this.key.position.set(l.keyX, l.keyY, l.keyZ);
    this.fill.intensity = l.fillIntensity;
    this.fill.color.set(l.fillColor);
    this.fill.position.set(-5, 1, 3);
    this.rim.intensity = l.rimIntensity;
    this.rim.position.set(l.rimX, l.rimY, l.rimZ);
    this.scene.environmentIntensity = l.envIntensity;
    this.renderer.toneMappingExposure = l.exposure;
  }

  setBloom(p: BloomSettings): void {
    this.bloom.enabled = p.bloom;
    this.bloom.threshold = p.bloomThreshold;
    this.bloom.strength = p.bloomStrength;
    this.bloom.radius = p.bloomRadius;
  }

  resize(): void {
    const w = innerWidth;
    const h = innerHeight;
    this.camera.aspect = w / h;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(w, h);
    this.composer.setSize(w, h);
  }

  render(): void {
    this.composer.render();
  }
}
```

```ts
import * as THREE from 'three';
import type { EyeSettings } from './config.ts';

export type SkinKind = 'map' | 'emissive' | 'rough';

export interface SkinInput {
  eyes: EyeSettings;
  tint: string;
  emissiveBoost: number;
  bodyRoughness: number;
  eyeGlow: number;
}

// Front of the sphere is u = 0.25, equator v = 0.5; near the equator a canvas circle is a sphere circle.
const W = 1024;
const H = 512;

const scaleHex = (hex: string, k: number): string =>
  '#' + new THREE.Color(hex).multiplyScalar(k).getHexString();

export function makeSkin(kind: SkinKind, s: SkinInput): THREE.CanvasTexture {
  const c = document.createElement('canvas');
  c.width = W;
  c.height = H;
  const g = c.getContext('2d')!;
  const e = s.eyes;
  const y = e.height * H;
  const xs = [(0.25 - e.spacing) * W, (0.25 + e.spacing) * W];

  if (kind === 'map') g.fillStyle = s.tint;
  else if (kind === 'emissive') g.fillStyle = scaleHex(s.tint, s.emissiveBoost);
  else g.fillStyle = '#000';
  g.fillRect(0, 0, W, H);

  xs.forEach((x, i) => {
    const side = i === 0 ? -1 : 1;
    g.save();
    g.translate(x, y);
    g.rotate(side * e.tilt);
    // Lids clip the ellipse: squint lowers the upper lid (1 = shut), lowerLid raises the lower one.
    const lid = e.squint * e.tall * 2;
    const lower = e.lowerLid * e.tall * 2;
    g.beginPath();
    g.rect(-e.width * 2, -e.tall - 2 + lid, e.width * 4, e.tall * 2 + 4 - lid - lower);
    g.clip();
    g.fillStyle = kind === 'map' ? e.color : kind === 'emissive' ? '#000000' : '#e0e0e0';
    g.beginPath();
    g.ellipse(0, 0, e.width, e.tall, 0, 0, Math.PI * 2);
    g.fill();
    if (e.highlight && kind !== 'rough') {
      g.fillStyle = kind === 'map' ? e.hlColor : scaleHex(e.hlColor, s.eyeGlow);
      g.beginPath();
      g.ellipse(e.hlOffsetX, e.hlOffsetY, e.hlSize, e.hlTall, 0, 0, Math.PI * 2);
      g.fill();
    }
    g.restore();
  });

  if (kind === 'rough') {
    // Roughness lives in the red channel; pixels the eye pass painted light are the eyes.
    const img = g.getImageData(0, 0, W, H);
    const d = img.data;
    const bodyR = Math.round(s.bodyRoughness * 255);
    const eyeR = Math.round(e.roughness * 255);
    for (let i = 0; i < d.length; i += 4) {
      const r = d[i] > 100 ? eyeR : bodyR;
      d[i] = d[i + 1] = d[i + 2] = r;
      d[i + 3] = 255;
    }
    g.putImageData(img, 0, 0);
  }

  const t = new THREE.CanvasTexture(c);
  if (kind !== 'rough') t.colorSpace = THREE.SRGBColorSpace;
  t.anisotropy = 8;
  return t;
}
```

- [ ] **Step 8: Write `src/character/neutrino.ts`**

Ported from the tuner's body, rim shell, halo and `apply()`. Flavor change tweens the material color (D-011) and redraws the skin once so its base tint matches, exactly as the tuner draws it.

```ts
import * as THREE from 'three';
import { gsap } from 'gsap';
import type { CharacterConfig, EyeSettings, Flavor } from './config.ts';
import { makeSkin } from './skin.ts';
import { REACTIONS, type ReactionName } from './reactions.ts';
import { TINTS } from './tints.ts';

export class Neutrino {
  readonly object = new THREE.Group();
  private readonly config: CharacterConfig;
  private readonly geometry = new THREE.SphereGeometry(1, 128, 96);
  private readonly material = new THREE.MeshPhysicalMaterial({
    transparent: true, metalness: 0, roughness: 1, emissive: new THREE.Color(0x000000),
  });
  private readonly body: THREE.Mesh;
  private readonly rimMaterial = new THREE.MeshBasicMaterial({
    transparent: true, blending: THREE.AdditiveBlending, side: THREE.BackSide, depthWrite: false,
  });
  private readonly rimShell: THREE.Mesh;
  private readonly halo: THREE.Sprite;
  private flavor: Flavor;
  private eyeOverride: Partial<EyeSettings> = {};
  private reaction: gsap.core.Timeline | null = null;
  private skinDrawn = false;

  constructor(config: CharacterConfig) {
    this.config = config;
    this.flavor = config.flavor;
    this.body = new THREE.Mesh(this.geometry, this.material);
    this.rimShell = new THREE.Mesh(this.geometry, this.rimMaterial);
    this.halo = new THREE.Sprite(new THREE.SpriteMaterial({
      map: makeHaloTexture(), transparent: true, blending: THREE.AdditiveBlending, depthWrite: false,
    }));
    this.halo.position.z = -0.6;
    this.body.add(this.rimShell, this.halo);
    this.object.add(this.body);
    this.applyBody();
    this.applyGlow();
    this.redrawSkin();
  }

  get currentFlavor(): Flavor {
    return this.flavor;
  }

  setFlavor(flavor: Flavor, seconds = 0.6): void {
    this.flavor = flavor;
    this.redrawSkin();
    this.applyGlow();
    gsap.to(this.material.color, { ...this.tintTarget(), duration: seconds });
  }

  react(name: ReactionName): void {
    const r = REACTIONS[name];
    const { scaleX, scaleY } = this.config.body;
    this.reaction?.kill();
    this.eyeOverride = r.eyes;
    this.redrawSkin();
    this.reaction = gsap.timeline({
      onComplete: () => {
        this.eyeOverride = {};
        this.redrawSkin();
        this.reaction = null;
      },
    })
      .to(this.body.scale, { x: scaleX * r.squash.x, y: scaleY * r.squash.y, duration: r.seconds * 0.35, ease: 'power2.out' })
      .to(this.body.scale, { x: scaleX, y: scaleY, duration: r.seconds * 0.65, ease: 'elastic.out(1, 0.45)' });
  }

  moveTo(x: number, y: number, z: number, seconds = 1): gsap.core.Tween {
    return gsap.to(this.object.position, { x, y, z, duration: seconds, ease: 'power2.inOut' });
  }

  dispose(): void {
    this.reaction?.kill();
    this.geometry.dispose();
    this.material.map?.dispose();
    this.material.roughnessMap?.dispose();
    this.material.emissiveMap?.dispose();
    this.material.dispose();
    this.rimMaterial.dispose();
    this.halo.material.map?.dispose();
    this.halo.material.dispose();
  }

  private tintTarget(): { r: number; g: number; b: number } {
    const c = new THREE.Color('#ffffff').lerp(new THREE.Color(TINTS[this.flavor]), this.config.body.colorDepth);
    return { r: c.r, g: c.g, b: c.b };
  }

  private applyBody(): void {
    const b = this.config.body;
    const m = this.material;
    this.body.scale.set(b.scaleX, b.scaleY, 1);
    const { r, g, b: blue } = this.tintTarget();
    m.color.setRGB(r, g, blue);
    m.opacity = b.opacity;
    m.clearcoat = b.clearcoat;
    m.clearcoatRoughness = b.clearcoatRoughness;
    m.sheen = b.sheen;
    m.sheenColor.set(b.sheenColor);
    m.emissive.set('#ffffff');
    m.emissiveIntensity = 1;
  }

  private applyGlow(): void {
    const p = this.config.post;
    const tint = TINTS[this.flavor];
    this.rimShell.visible = p.rimShell;
    this.rimMaterial.color.set(tint);
    this.rimMaterial.opacity = p.rimOpacity;
    this.rimShell.scale.setScalar(p.rimScale);
    this.halo.visible = p.halo;
    this.halo.material.color.set(tint);
    this.halo.material.opacity = p.haloOpacity;
    this.halo.scale.setScalar(p.haloSize);
  }

  private redrawSkin(): void {
    const input = {
      eyes: { ...this.config.eyes, ...this.eyeOverride },
      tint: TINTS[this.flavor],
      emissiveBoost: this.config.body.emissiveBoost,
      bodyRoughness: this.config.body.roughness,
      eyeGlow: this.config.post.eyeGlow,
    };
    const m = this.material;
    m.map?.dispose();
    m.roughnessMap?.dispose();
    m.emissiveMap?.dispose();
    m.map = makeSkin('map', input);
    m.roughnessMap = makeSkin('rough', input);
    m.emissiveMap = makeSkin('emissive', input);
    if (!this.skinDrawn) {
      // The shader only needs recompiling when the maps first appear.
      m.needsUpdate = true;
      this.skinDrawn = true;
    }
  }
}

function makeHaloTexture(): THREE.CanvasTexture {
  const c = document.createElement('canvas');
  c.width = c.height = 256;
  const g = c.getContext('2d')!;
  const gr = g.createRadialGradient(128, 128, 0, 128, 128, 128);
  gr.addColorStop(0, 'rgba(255,255,255,1)');
  gr.addColorStop(0.35, 'rgba(255,255,255,0.35)');
  gr.addColorStop(1, 'rgba(255,255,255,0)');
  g.fillStyle = gr;
  g.fillRect(0, 0, 256, 256);
  return new THREE.CanvasTexture(c);
}
```

- [ ] **Step 5: Write `src/levels/manager.ts`**

```ts
import type { Level, LevelContext } from './types.ts';

export type LevelFactory = () => Level;

export class LevelManager {
  private readonly factories: LevelFactory[];
  private readonly base: Omit<LevelContext, 'done'>;
  private current: Level | null = null;
  private index = -1;

  constructor(factories: LevelFactory[], base: Omit<LevelContext, 'done'>) {
    this.factories = factories;
    this.base = base;
  }

  get level(): number {
    return this.index + 1;
  }

  get finished(): boolean {
    return this.current === null && this.index >= this.factories.length;
  }

  start(at = 1): void {
    this.enter(at - 1);
  }

  next(): void {
    this.current?.next();
  }

  update(dt: number): void {
    this.current?.update(dt);
  }

  private enter(i: number): void {
    this.current?.exit();
    this.current = null;
    this.index = i;
    if (i >= this.factories.length) return;
    const level = this.factories[i]!();
    this.current = level;
    level.enter({
      ...this.base,
      done: () => {
        if (this.current === level) this.enter(i + 1);
      },
    });
  }
}
```

```ts
export interface Flags {

export function parseFlags(search: string, levelCount: number): Flags {
  const params = new URLSearchParams(search);
  const raw = Number(params.get('level'));
  const level = Number.isInteger(raw) && raw >= 1 && raw <= levelCount ? raw : 1;
  return { level, debug: params.has('debug') };
}
```

```js
#!/usr/bin/env node
// D-002, D-003: every F-ID used in src/ must exist in docs/FACTS.md, and the mirror must match the doc.
import { readFileSync, readdirSync, statSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { pathToFileURL } from 'node:url';

const [srcDir = 'src', factsDoc = 'docs/FACTS.md', mirror = 'src/content/facts.ts'] = process.argv.slice(2);

const documented = new Set([...readFileSync(factsDoc, 'utf8').matchAll(/^### (F-\d{2})\b/gm)].map((m) => m[1]));

const referenced = new Map();
for (const file of walk(srcDir)) {
  for (const m of readFileSync(file, 'utf8').matchAll(/\bF-\d{2}\b/g)) {
    if (!referenced.has(m[0])) referenced.set(m[0], file);
  }
}

const { FACTS } = await import(pathToFileURL(resolve(mirror)).href);
const mirrored = new Set(Object.keys(FACTS));

const problems = [];
for (const [id, file] of referenced) if (!documented.has(id)) problems.push(`${id} is referenced in ${file} but has no entry in ${factsDoc}`);
for (const id of documented) if (!mirrored.has(id)) problems.push(`${id} is in ${factsDoc} but missing from ${mirror}`);
for (const id of mirrored) if (!documented.has(id)) problems.push(`${id} is in ${mirror} but not in ${factsDoc}`);

if (problems.length > 0) {
  console.error(problems.join('\n'));
  process.exit(1);
}
console.log(`check-facts: ${documented.size} facts documented, ${referenced.size} referenced in ${srcDir}, mirror in sync`);

function walk(dir, out = []) {
  for (const name of readdirSync(dir)) {
    const p = join(dir, name);
    if (statSync(p).isDirectory()) walk(p, out);
    else out.push(p);
  }
  return out;
}
```

```ts
import { GUI } from 'three/addons/libs/lil-gui.module.min.js';
import Stats from 'three/addons/libs/stats.module.js';
import type { LevelManager } from '../levels/manager.ts';
import type { Neutrino } from '../character/neutrino.ts';
import type { Flavor } from '../character/config.ts';
import { REACTION_NAMES, type ReactionName } from '../character/reactions.ts';

export interface DebugPanel {
  update(): void;
}

const FLAVORS: Flavor[] = ['electron', 'muon', 'tau'];

// D-017: mounted only when the URL carries ?debug.
export function mountDebug(manager: LevelManager, neutrino: Neutrino): DebugPanel {
  const stats = new Stats();
  document.body.append(stats.dom);

  const gui = new GUI({ title: 'Ghost Particle debug' });
  const state = { level: manager.level, reaction: 'wiggle' as ReactionName, flavor: neutrino.currentFlavor };
  const levelDisplay = gui.add(state, 'level').disable();
  gui.add({ next: () => manager.next() }, 'next').name('next beat');
  gui.add(state, 'reaction', REACTION_NAMES);
  gui.add({ react: () => neutrino.react(state.reaction) }, 'react');
  gui.add(state, 'flavor', FLAVORS).onChange((flavor: Flavor) => neutrino.setFlavor(flavor));

  return {
    update() {
      stats.update();
      if (state.level !== manager.level) {
        state.level = manager.level;
        levelDisplay.updateDisplay();
      }
    },
  };
}
```
