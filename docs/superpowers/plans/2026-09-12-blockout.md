# Blockout Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. No per-task review agents; `npm run check` before every commit.

**Goal:** Every beat of `docs/BEATS.md` playable end to end in grey primitives, with the real camera paths, captions, inputs, and timings, so the game can be played before any polish.

**Architecture:** Levels stay one file each and become async scripts over a small beat sequencer (`wait`, `card`, `key`, `until`) that the debug hook can step. A DOM HUD (D-016) owns cards, prompts, meters, the through-line counter, the tile grid, the E/M buttons, the chart, and the credits. A camera rig moves `ctx.camera` along each level's path. Primitives are one grey `MeshStandardMaterial` on three.js geometry; nothing else is styled.

**Tech Stack:** TypeScript 6, Vite 8, three 0.186, Playwright 1.63. No new dependencies.

**Spec:** `docs/BEATS.md` (the "Blockout" and "Target lengths" sections bind this plan), `docs/SPEC.md`, `docs/DECISIONS.md`, `docs/FACTS.md`.

## Global Constraints

- Blockout scope (BEATS.md "Blockout"): build every row's *Player sees*, *Player does*, *Caption*, *Facts*. Ignore *Neutrino* and *Cue*: the neutrino is a plain translucent grey sphere, no animation, no sound. Grey primitives only. No materials, lighting, assets, or bloom changes. The stage (`src/render/stage.ts`) is not edited.
- Captions verbatim from BEATS.md, at most 25 words, tagged with their F-IDs in code (`card(text, ['F-03'])`). Pure instruction cards carry no tag. `npm run check-facts` must stay green.
- Input map (BEATS.md): Space = hold in level 1, continue elsewhere. Left/Right = steer in level 3, move the grid highlight in level 4. Hold Right = fast-forward in level 4. M = measure in level 4. E / M = electron / muon in level 5. Clicks work on every DOM control; nothing requires them.
- Rail rule (levels 3 and 4): while a card is up the rail runs at one third speed; resumes on Space.
- Target lengths (BEATS.md): 1: 20 s, 2: 30 s, 3: 60 s, 4: 75 s, 5: 90 s, 6: 60 s. Cards do not auto-advance; budget about 5 s of reading per card.
- Flavor tints (D-011): electron `#5ee3ff`, muon `#b58cff`, tau `#ff9f6b`. In the blockout the sphere's flat colour and the HUD flavor icon take the tint; nothing else is coloured.
- Levels own their three.js objects and dispose them in `exit()` (D-018). `hud.clear()` in `exit()` removes every HUD element except the counter.
- tsconfig: strict, `verbatimModuleSyntax`, `erasableSyntaxOnly` (no enums, no parameter properties), `.ts` extensions on imports, `import type` for types.
- Level 7 "Want more?" buttons (2.4, 3.10, 5.12) are not built.
- Level 6 uses placeholder data (A-02 is not downloaded in the blockout) and says so on screen.
- Commit trailer: `Co-Authored-By: <model name> <noreply@anthropic.com>` naming the authoring model.

// src/content/flavors.ts
export type Flavor = 'electron' | 'muon' | 'tau';
export const FLAVORS: readonly Flavor[] = ['electron', 'muon', 'tau'];
export const TINTS: Record<Flavor, string> = { electron: '#5ee3ff', muon: '#b58cff', tau: '#ff9f6b' };

// src/input/keys.ts
export class Keys {
  constructor(target?: EventTarget);           // default window; preventDefault on Space and arrows
  isDown(code: string): boolean;               // KeyboardEvent.code, e.g. 'Space', 'ArrowRight', 'KeyM'
  onPress(code: string, fn: () => void): () => void;   // once per keydown (no auto-repeat); returns unsubscribe
  onRelease(code: string, fn: () => void): () => void;
  press(code: string): void;                   // synthetic keydown + keyup (tests, ghost hook)
  dispose(): void;
}

// src/hud/counter.ts
export const RATE_PER_SECOND = 6e10 * 5400;     // F-32 planning rate: Φ (F-07) × A
export function formatCount(n: number): string; // "about 320 trillion": 3 significant figures + word scale
                                                // scales: thousand, million, billion, trillion, quadrillion, quintillion
export class Counter {
  constructor(root: HTMLElement);               // creates its own element inside root, hidden
  start(): void;                                // clock starts at first call; later calls are no-ops
  stop(): void;
  readonly running: boolean;
  value(): number;                              // RATE_PER_SECOND × seconds since start (0 before start)
  setVisible(v: boolean): void;                 // top-right corner: "Neutrinos through you since you started: …"
  freezeLarge(): void;                          // stop, move to centre, grow (6.5)
  update(): void;                               // refreshes the DOM at most 10× per second
}

// src/hud/hud.ts
export interface GridTile { row: string; name: string; label: string }
export interface GridHandle {
  onPick(fn: (index: number) => void): void;    // arrows move the highlight, Space or click picks
  wiggle(index: number, seconds: number): void; // CSS shake + label enlarged for that long (4.5)
  setLabel(index: number, label: string): void;
  mark(index: number): void;                    // filled highlight for the right tile
  close(): void;
}
export interface ChartPoint { x: number; y: number; err: number }
export interface ChartHandle {
  draw(seconds: number): Promise<void>;         // reveals points left to right over that long
  curve(fn: (x: number) => number, seconds: number): Promise<void>;  // draws a line over the points
  close(): void;
}
export class Hud {
  readonly counter: Counter;
  constructor(root: HTMLElement, keys: Keys);
  readonly cardUp: boolean;                     // true while a card() promise is pending (rail rule)
  card(text: string, facts?: string[], opts?: { small?: boolean }): Promise<void>;
                                                // bottom-centre card, fact tags small under the text,
                                                // "Space ▸" hint; resolves on Space or click; one at a time
  note(text: string | null, facts?: string[]): void;   // a card with no Space hint that stays until cleared
                                                // (used while the grid owns Space in level 4)
  prompt(text: string | null): void;            // centre key prompt ("Hold Space")
  title(text: string | null): void;             // large title above centre
  big(text: string, seconds: number): void;     // one-shot large centred text (3.9)
  meter(fraction: number | null): void;         // thin bar under centre (level 1 charge meter)
  readout(text: string | null): void;           // top-centre text (level 4 travel meter)
  sub(lines: string[]): void;                   // small lines under the counter (tallies)
  flavor(f: Flavor | null): void;               // flavor icon under the counter, tinted (D-011)
  flash(seconds: number): void;                 // white overlay fading out (1.5)
  fade(to: 0 | 1, seconds: number): Promise<void>;  // black overlay: 1 = black, 0 = clear
  grid(tiles: GridTile[], columns: number): GridHandle;
  buttons(items: { key: string; label: string }[], onPress: (key: string) => void): { close(): void };
                                                // "E  Electron" style buttons; key press or click
  reveal(text: string, seconds: number): void;  // one line above the buttons for that long (5.9)
  chart(points: ChartPoint[], opts: { xLabel: string; yLabel: string; credit: string; note: string }): ChartHandle;
                                                // SVG on the left 65% of the screen, error bars per point
  credits(lines: string[], onPlayAgain: () => void): { close(): void };
                                                // scrolling list; "Play again" button visible at once
  clear(): void;                                // removes everything except the counter
}

// src/levels/beats.ts
export interface CardHost { card(text: string, facts?: string[], opts?: { small?: boolean }): Promise<void> }
export interface PressHost { onPress(code: string, fn: () => void): () => void }
export class Beats {
  constructor(host: CardHost, keys: PressHost);
  wait(seconds: number): Promise<void>;         // counts update(dt); next() resolves early
  until(pred: () => boolean): Promise<void>;    // checked each update(); next() resolves early
  card(text: string, facts?: string[], opts?: { small?: boolean }): Promise<void>;  // host.card; next() resolves
  key(code: string): Promise<void>;             // resolves on that key's press; next() resolves
  next(): void;                                 // resolves whichever step is pending
  cancel(): void;                               // after this every pending and future step never resolves
  readonly cancelled: boolean;
  update(dt: number): void;
}

// src/levels/types.ts
export interface LevelContext {
  scene: Scene; camera: PerspectiveCamera; hud: Hud; keys: Keys; rig: CameraRig; done: () => void;
}
export interface Level { readonly id: number; enter(ctx): void; update(dt): void; exit(): void; next(): void }
// next() now advances one beat (Beats.next). Skipping a level is LevelManager.skip().

// src/levels/scripted.ts
export interface Scripted {
  ctx: LevelContext; b: Beats; hud: Hud; keys: Keys; rig: CameraRig; scene: Scene;
  group: Group;                                 // added to the scene on enter, disposed on exit
  time: number;                                 // seconds since enter
  onUpdate(fn: (dt: number) => void): void;     // per-frame hook, cleared on exit
}
export function scriptedLevel(id: number, script: (s: Scripted) => Promise<void>): LevelFactory;
// enter: builds Beats, runs script; when the script resolves, ctx.done(). exit: b.cancel(), hud.clear(),
// scene.fog = null, disposeGroup(group), unsubscribe keys. next: b.next().

// src/render/rig.ts
export class CameraRig {
  constructor(camera: PerspectiveCamera);
  set(pos: Vec3Like, look: Vec3Like): void;
  moveTo(pos: Vec3Like, look: Vec3Like, seconds: number): Promise<void>;   // eased; resolves when done
  shake(amount: number): void;                  // per-frame jitter amplitude; 0 disables; no-op under reduced motion
  update(dt: number): void;
}
export type Vec3Like = { x: number; y: number; z: number } | [number, number, number];
export function prefersReducedMotion(): boolean;

// src/render/prims.ts   (one shared grey MeshStandardMaterial per call, transparent when opacity < 1)
export const GREY = '#9aa4ad';
export interface PrimOpts { opacity?: number; color?: string }
export function sphere(radius: number, o?: PrimOpts): Mesh;
export function box(w: number, h: number, d: number, o?: PrimOpts): Mesh;
export function plane(w: number, h: number, o?: PrimOpts): Mesh;       // DoubleSide
export function ring(radius: number, tube: number, o?: PrimOpts): Mesh; // TorusGeometry

export function disposeGroup(group: Object3D): void;         // geometries and materials, then removes children
